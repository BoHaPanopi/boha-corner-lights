cmake_minimum_required(VERSION 3.20)
project(BoHaCornerLights VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DONOR "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
add_library(minhook STATIC ${DONOR}/minhook/src/buffer.c ${DONOR}/minhook/src/hook.c
 ${DONOR}/minhook/src/trampoline.c ${DONOR}/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC ${DONOR}/minhook/include)
add_library(BoHaCornerLights SHARED src/main.cpp src/exports.def)
target_include_directories(BoHaCornerLights PRIVATE ${DONOR}/scssdk/include)
target_compile_definitions(BoHaCornerLights PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(BoHaCornerLights PRIVATE minhook)
add_executable(controller_test src/test.cpp)
target_compile_options(controller_test PRIVATE /UNDEBUG)
enable_testing()
add_test(NAME corner_controller COMMAND controller_test)
